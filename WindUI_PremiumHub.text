-- Palette button: a compact scrollable theme selector rather than a moon toggle.
local ThemeMenuGui
local function OpenThemeMenu()
    if ThemeMenuGui then ThemeMenuGui:Destroy(); ThemeMenuGui = nil; return end
    ThemeMenuGui = Instance.new("ScreenGui")
    ThemeMenuGui.Name = "NexusThemeSelector"
    ThemeMenuGui.ResetOnSpawn = false
    ThemeMenuGui.DisplayOrder = 1000
    ThemeMenuGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
    ThemeMenuGui.Parent = (gethui and gethui()) or game:GetService("CoreGui")

    local menu = Instance.new("Frame")
    menu.AnchorPoint = Vector2.new(1, 0)
    menu.Position = UDim2.new(1, -26, 0, 66)
    menu.Size = UDim2.fromOffset(230, 260)
    menu.BackgroundColor3 = Color3.fromRGB(18, 18, 28)
    menu.BorderSizePixel = 0
    menu.Parent = ThemeMenuGui
    Instance.new("UICorner", menu).CornerRadius = UDim.new(0, 10)
    local outline = Instance.new("UIStroke", menu)
    outline.Color = Color3.fromRGB(80, 70, 120)
    outline.Transparency = .2

    local title = Instance.new("TextLabel")
    title.BackgroundTransparency = 1
    title.Position = UDim2.fromOffset(12, 8)
    title.Size = UDim2.new(1, -24, 0, 21)
    title.Text = "THEME PALETTE"
    title.TextColor3 = Color3.fromRGB(240, 238, 255)
    title.Font = Enum.Font.GothamBold
    title.TextSize = 12
    title.TextXAlignment = Enum.TextXAlignment.Left
    title.Parent = menu

    local list = Instance.new("ScrollingFrame")
    list.BackgroundTransparency = 1
    list.BorderSizePixel = 0
    list.Position = UDim2.fromOffset(8, 34)
    list.Size = UDim2.new(1, -16, 1, -42)
    list.CanvasSize = UDim2.new()
    list.AutomaticCanvasSize = Enum.AutomaticSize.Y
    list.ScrollBarThickness = 3
    list.ScrollBarImageColor3 = Color3.fromRGB(120, 95, 220)
    list.Parent = menu
    local layout = Instance.new("UIListLayout", list)
    layout.Padding = UDim.new(0, 4)
    layout.SortOrder = Enum.SortOrder.LayoutOrder

    for _, data in ipairs(FluentProThemes) do
        local name, hex = data[1], data[2]
        local choice = Instance.new("TextButton")
        choice.Size = UDim2.new(1, 0, 0, 30)
        choice.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
        choice.BackgroundTransparency = .94
        choice.BorderSizePixel = 0
        choice.Text = ""
        choice.AutoButtonColor = false
        choice.Parent = list
        Instance.new("UICorner", choice).CornerRadius = UDim.new(0, 6)
        local swatch = Instance.new("Frame")
        swatch.Size = UDim2.fromOffset(14, 14)
        swatch.Position = UDim2.fromOffset(8, 8)
        swatch.BackgroundColor3 = Color3.fromHex(hex)
        swatch.BorderSizePixel = 0
        swatch.Parent = choice
        Instance.new("UICorner", swatch).CornerRadius = UDim.new(1, 0)
        local choiceText = Instance.new("TextLabel")
        choiceText.BackgroundTransparency = 1
        choiceText.Position = UDim2.fromOffset(30, 0)
        choiceText.Size = UDim2.new(1, -36, 1, 0)
        choiceText.Text = name
        choiceText.TextColor3 = Color3.fromRGB(235, 233, 245)
        choiceText.Font = Enum.Font.GothamMedium
        choiceText.TextSize = 11
        choiceText.TextXAlignment = Enum.TextXAlignment.Left
        choiceText.Parent = choice
        choice.MouseEnter:Connect(function() choice.BackgroundTransparency = .86 end)
        choice.MouseLeave:Connect(function() choice.BackgroundTransparency = .94 end)
        choice.MouseButton1Click:Connect(function()
            WindUI:SetTheme(name)
            WindUI:Notify({Title = "Theme applied", Content = name, Icon = "palette", Duration = 2})
            ThemeMenuGui:Destroy(); ThemeMenuGui = nil
        end)
    end
end
Window:CreateTopbarButton("theme-palette", "palette", OpenThemeMenu, 990)
